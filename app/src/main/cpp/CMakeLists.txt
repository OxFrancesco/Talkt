cmake_minimum_required(VERSION 3.22)
project(talkt-parakeet LANGUAGES C CXX)

include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_LIB_OUTPUT_PREFIX "speech-" CACHE STRING "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_INSTALL OFF CACHE BOOL "" FORCE)
set(PARAKEET_OPENMP OFF CACHE BOOL "" FORCE)
set(PARAKEET_FLASH_ATTN OFF CACHE BOOL "" FORCE)
set(PARAKEET_F16_ACTIVATIONS ON CACHE BOOL "" FORCE)
option(TALKT_HEXAGON "Build Qualcomm Hexagon backend using the installed SDK" OFF)
set(GGML_HEXAGON ${TALKT_HEXAGON} CACHE BOOL "" FORCE)

FetchContent_Declare(opencl_headers URL https://github.com/KhronosGroup/OpenCL-Headers/archive/8a97ebc88daa3495d6f57ec10bb515224400186f.tar.gz)
FetchContent_GetProperties(opencl_headers)
if(NOT opencl_headers_POPULATED)
    FetchContent_Populate(opencl_headers)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(OpenCL::Headers INTERFACE IMPORTED)
set_target_properties(OpenCL::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${opencl_headers_SOURCE_DIR}")
set(OPENCL_DISPATCH "${CMAKE_CURRENT_BINARY_DIR}/opencl_dispatch.cpp")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/opencl_dispatch.py"
    "${opencl_headers_SOURCE_DIR}/CL/cl.h" "${OPENCL_DISPATCH}"
    COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/opencl_dispatch.py")
add_library(OpenCL STATIC "${OPENCL_DISPATCH}")
target_link_libraries(OpenCL PUBLIC OpenCL::Headers PRIVATE dl log)
install(TARGETS OpenCL EXPORT ggml-targets ARCHIVE DESTINATION lib)
set(OpenCL_LIBRARY OpenCL CACHE STRING "" FORCE)
set(OpenCL_INCLUDE_DIR "${opencl_headers_SOURCE_DIR}" CACHE PATH "" FORCE)

FetchContent_Declare(speech_ggml URL https://github.com/tetherto/qvac-ext-ggml/archive/0588836eee661126700545c4e7a4c8a133dc39b1.tar.gz)
FetchContent_MakeAvailable(speech_ggml)
FetchContent_Declare(qvac_speech URL https://github.com/tetherto/qvac-fabric-speech.cpp/archive/588b8e98db1ae8b8ecb325e114b42d7901a70811.tar.gz)
FetchContent_GetProperties(qvac_speech)
if(NOT qvac_speech_POPULATED)
    FetchContent_Populate(qvac_speech)
endif()
add_subdirectory("${qvac_speech_SOURCE_DIR}/engines/parakeet" "${qvac_speech_BINARY_DIR}/parakeet")

add_library(talkt-parakeet SHARED parakeet_jni.cpp)
target_link_libraries(talkt-parakeet PRIVATE qvac::parakeet log OpenCL::Headers)
target_compile_definitions(talkt-parakeet PRIVATE TALKT_HEXAGON=$<BOOL:${TALKT_HEXAGON}>)
target_link_options(talkt-parakeet PRIVATE "-Wl,--exclude-libs,ALL")

set(TALKT_LICENSE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/generated/parakeet-assets/licenses")
file(MAKE_DIRECTORY "${TALKT_LICENSE_DIR}")
configure_file("${qvac_speech_SOURCE_DIR}/engines/parakeet/LICENSE" "${TALKT_LICENSE_DIR}/QVAC-Parakeet-Apache-2.0.txt" COPYONLY)
configure_file("${speech_ggml_SOURCE_DIR}/LICENSE" "${TALKT_LICENSE_DIR}/QVAC-GGML-MIT.txt" COPYONLY)
configure_file("${opencl_headers_SOURCE_DIR}/LICENSE" "${TALKT_LICENSE_DIR}/OpenCL-Headers-Apache-2.0.txt" COPYONLY)

if(TALKT_HEXAGON)
    add_dependencies(talkt-parakeet htp-v81)
    add_custom_command(TARGET talkt-parakeet POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/generated/hexagon/arm64-v8a"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${speech_ggml_BINARY_DIR}/src/ggml-hexagon/libggml-htp-v81.so"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/generated/hexagon/arm64-v8a/libggml-htp-v81.so")
endif()
